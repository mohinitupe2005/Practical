#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>

#define MAX 20

void sortArray(int arr[], int n) {
    int i, j, temp;
    for (i = 0; i < n-1; i++) {
        for (j = i+1; j < n; j++) {
            if (arr[i] > arr[j]) {
                temp = arr[i]; arr[i] = arr[j]; arr[j] = temp;
            }
        }
    }
}

int main() {
    int n, i;
    int arr[MAX];

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for (i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    sortArray(arr, n);
    printf("Parent: Sorted array: ");
    for (i = 0; i < n; i++) printf("%d ", arr[i]);
    printf("\n");

    pid_t pid = fork();

    if (pid < 0) {
        perror("fork failed");
        exit(1);
    } 
    else if (pid == 0) {
        char *args[MAX + 2];
        args[0] = "./child";
        char buffer[20];

        for (i = 0; i < n; i++) {
            sprintf(buffer, "%d", arr[i]);
            args[i+1] = strdup(buffer);
        }
        args[n+1] = NULL;

        if (execve("./child", args, NULL) == -1) {
            perror("execve failed");
            exit(1);
        }
    } 
    else {
        wait(NULL);
        printf("Parent: Child finished execution.\n");
    }

    return 0;
}
