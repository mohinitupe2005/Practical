#include <stdio.h>
#include <stdlib.h>

int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("No array elements received.\n");
        return 1;
    }

    printf("Child: Array in reverse order: ");
    for (int i = argc-1; i >= 1; i--) {
        printf("%s ", argv[i]);
    }
    printf("\n");

    return 0;
}
