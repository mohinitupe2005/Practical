#include <stdio.h>

int main() {
    int pages[30], frames[10], lastUsed[10];
    int n, f, i, j, k, pageFaults = 0, flag, lruIndex, minTime;

    printf("Enter number of pages: ");
    scanf("%d", &n);

    printf("Enter page reference string:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &pages[i]);

    printf("Enter number of frames (minimum 3): ");
    scanf("%d", &f);
    if (f < 3) {
        printf("Frame size must be at least 3!\n");
        return 0;
    }

    // Initialize frames and lastUsed
    for (i = 0; i < f; i++) {
        frames[i] = -1;
        lastUsed[i] = -1;
    }

    printf("\nPage\tFrames\t\tPage Fault\n");
    printf("-------------------------------------\n");

    for (i = 0; i < n; i++) {
        flag = 0;

        // Check if page is already in frame
        for (j = 0; j < f; j++) {
            if (frames[j] == pages[i]) {
                flag = 1;  // Page hit
                lastUsed[j] = i;  // Update last used time
                break;
            }
        }

        // Page fault
        if (flag == 0) {
            pageFaults++;

            // Check if empty frame is available
            int placed = 0;
            for (j = 0; j < f; j++) {
                if (frames[j] == -1) {
                    frames[j] = pages[i];
                    lastUsed[j] = i;
                    placed = 1;
                    break;
                }
            }

            // If no empty frame, replace least recently used
            if (placed == 0) {
                minTime = lastUsed[0];
                lruIndex = 0;
                for (j = 1; j < f; j++) {
                    if (lastUsed[j] < minTime) {
                        minTime = lastUsed[j];
                        lruIndex = j;
                    }
                }
                frames[lruIndex] = pages[i];
                lastUsed[lruIndex] = i;
            }
        }

        // Print current frame status
        printf("%d\t", pages[i]);
        for (j = 0; j < f; j++) {
            if (frames[j] != -1)
                printf("%d ", frames[j]);
            else
                printf("- ");
        }
        if (flag == 0)
            printf("\t\tYes\n");
        else
            printf("\t\tNo\n");
    }

    printf("-------------------------------------\n");
    printf("Total Page Faults = %d\n", pageFaults);

    return 0;
}
